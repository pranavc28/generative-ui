import React, { useState, useEffect, useMemo, memo, createContext, useContext, useCallback } from 'react';
import ReactDOM from 'react-dom';

interface Booking {
  id: string;
  pickupLocation: string;
  dropoffLocation: string;
  pickupTime: Date;
  customerName: string;
  customerPhone: string;
  status: 'pending' | 'assigned' | 'completed';
  driverId?: string;
}

interface Driver {
  id: string;
  name: string;
  isOnline: boolean;
}

interface AppState {
  bookings: Booking[];
  drivers: Driver[];
  selectedBookingId: string | null;
  selectedDriverId: string | null;
}

const initialState: AppState = {
  bookings: [],
  drivers: [
    { id: '1', name: 'John Doe', isOnline: false },
    { id: '2', name: 'Jane Smith', isOnline: false },
  ],
  selectedBookingId: null,
  selectedDriverId: null,
};

const AppContext = createContext<{ state: AppState; dispatch: React.Dispatch<Action> }>({
  state: initialState,
  dispatch: () => {},
});

type Action =
  | { type: 'ADD_BOOKING'; payload: Booking }
  | { type: 'ASSIGN_DRIVER'; payload: { bookingId: string; driverId: string } }
  | { type: 'TOGGLE_DRIVER_STATUS'; payload: string }
  | { type: 'SELECT_BOOKING'; payload: string | null }
  | { type: 'SELECT_DRIVER'; payload: string | null };

const appReducer = (state: AppState, action: Action): AppState => {
  switch (action.type) {
    case 'ADD_BOOKING':
      return { ...state, bookings: [...state.bookings, action.payload] };
    case 'ASSIGN_DRIVER':
      const updatedBookings = state.bookings.map((booking) =>
        booking.id === action.payload.bookingId
          ? { ...booking, status: 'assigned', driverId: action.payload.driverId }
          : booking
      );
      return { ...state, bookings: updatedBookings };
    case 'TOGGLE_DRIVER_STATUS':
      const updatedDrivers = state.drivers.map((driver) =>
        driver.id === action.payload ? { ...driver, isOnline: !driver.isOnline } : driver
      );
      return { ...state, drivers: updatedDrivers };
    case 'SELECT_BOOKING':
      return { ...state, selectedBookingId: action.payload };
    case 'SELECT_DRIVER':
      return { ...state, selectedDriverId: action.payload };
    default:
      return state;
  }
};

const AppProvider: React.FC<{ children: React.ReactNode }> = ({ children }) => {
  const [state, dispatch] = React.useReducer(appReducer, initialState);

  return (
    <AppContext.Provider value={{ state, dispatch }}>
      {children}
    </AppContext.Provider>
  );
};

const BookingCard = memo(({ booking }: { booking: Booking }) => {
  const { state, dispatch } = useAppContext();

  const availableDrivers = useMemo(() => state.drivers.filter((driver) => driver.isOnline), [state.drivers]);
  const [selectedDriver, setSelectedDriver] = useState<string | null>(null);

  const handleAssignDriver = useCallback(() => {
    if (selectedDriver) {
      dispatch({ type: 'ASSIGN_DRIVER', payload: { bookingId: booking.id, driverId: selectedDriver } });
      setSelectedDriver(null);
    }
  }, [dispatch, booking.id, selectedDriver]);

  return (
    <div className="bg-white p-6 rounded-lg shadow-md mb-4">
      <h3 className="text-xl font-bold mb-2">Booking #{booking.id}</h3>
      <p><strong>Pickup Location:</strong> {booking.pickupLocation}</p>
      <p><strong>Dropoff Location:</strong> {booking.dropoffLocation}</p>
      <p><strong>Pickup Time:</strong> {new Date(booking.pickupTime).toLocaleString()}</p>
      <p><strong>Customer Name:</strong> {booking.customerName}</p>
      <p><strong>Customer Phone:</strong> {booking.customerPhone}</p>
      <p><strong>Status:</strong> {booking.status}</p>
      {booking.status === 'pending' && (
        <div className="mt-4">
          <label htmlFor={`driver-${booking.id}`} className="block text-gray-700 font-bold mb-2">
            Assign Driver
          </label>
          <select
            id={`driver-${booking.id}`}
            value={selectedDriver || ''}
            onChange={(e) => setSelectedDriver(e.target.value)}
            className="shadow appearance-none border rounded w-full py-2 px-3 text-gray-700 leading-tight focus:outline-none focus:shadow-outline"
          >
            <option value="">Select a driver</option>
            {availableDrivers.map((driver) => (
              <option key={driver.id} value={driver.id}>
                {driver.name}
              </option>
            ))}
          </select>
          <button
            onClick={handleAssignDriver}
            className="mt-2 bg-green-500 hover:bg-green-700 text-white font-bold py-2 px-4 rounded focus:outline-none focus:shadow-outline"
            disabled={!selectedDriver}
          >
            Assign
          </button>
        </div>
      )}
    </div>
  );
});
