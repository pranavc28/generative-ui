import React, { useState, useEffect } from 'react';
import { ReactSVG } from 'react-svg';
import { useTheme } from './ThemeContext'; // Assuming a context for dark mode

interface Rank {
  id: number;
  name: string;
  score: number;
  country: string;
}

const mockData: Rank[] = [
  { id: 1, name: 'Alice', score: 5000, country: 'USA' },
  { id: 2, name: 'Bob', score: 4500, country: 'Canada' },
  { id: 3, name: 'Charlie', score: 4000, country: 'UK' },
  { id: 4, name: 'David', score: 3500, country: 'Australia' },
  { id: 5, name: 'Eve', score: 3000, country: 'Germany' },
];

const RankCard: React.FC<Rank> = ({ id, name, score, country }) => {
  const { isDarkMode } = useTheme();

  return (
    <div
      className={`p-4 flex items-center justify-between rounded-lg mb-4 ${
        isDarkMode ? 'bg-gray-800 text-white' : 'bg-white text-gray-800'
      } shadow-md`}
    >
      <div className="flex items-center">
        <div className="flex-shrink-0 w-12 h-12 bg-gray-200 rounded-full flex items-center justify-center mr-4">
          <span className="text-lg font-bold text-gray-600">{id}</span>
        </div>
        <div>
          <h3 className="text-lg font-medium">{name}</h3>
          <p className="text-sm text-gray-600">{country}</p>
        </div>
      </div>
      <div className="text-right">
        <p className="text-lg font-bold">{score}</p>
        <p className="text-sm text-gray-600">Points</p>
      </div>
    </div>
  );
};

const App: React.FC = () => {
  const [ranks, setRanks] = useState<Rank[]>(mockData);
  const { isDarkMode, toggleTheme } = useTheme();

  useEffect(() => {
    const interval = setInterval(() => {
      setRanks((prevRanks) =>
        prevRanks.map((rank) => ({
          ...rank,
          score: rank.score + Math.floor(Math.random() * 100),
        }))
      );
    }, 3000);

    return () => clearInterval(interval);
  }, []);

  return (
    <div
      className={`min-h-screen flex flex-col items-center justify-center p-6 ${
        isDarkMode ? 'bg-gray-900 text-white' : 'bg-gray-100 text-gray-900'
      } transition-colors duration-300`}
    >
      <header className="w-full max-w-4xl mb-8">
        <div className="flex items-center justify-between">
          <h1 className="text-3xl font-bold">Realtime Ranking List</h1>
          <button
            className="p-2 rounded-lg bg-blue-500 text-white hover:bg-blue-600 transition-colors duration-300"
            onClick={toggleTheme}
          >
            {isDarkMode ? (
              <svg
                xmlns="http://www.w3.org/2000/svg"
                viewBox="0 0 24 24"
                fill="currentColor"
                className="w-6 h-6"
              >
                <path d="M12 2L2 22h20L12 2z" />
              </svg>
            ) : (
              <svg
                xmlns="http://www.w3.org/2000/svg"
                viewBox="0 0 24 24"
                fill="currentColor"
                className="w-6 h-6"
              >
                <path d="M12 22a10 10 0 110-20 10 10 0 010 20zm0-2a8 8 0 100-16 8 8 0 000 16z" />
              </svg>
            )}
          </button>
        </div>
      </header>
      <main className="w-full max-w-4xl">
        {ranks
          .slice()
          .sort((a, b) => b.score - a.score)
          .map((rank) => (
            <RankCard key={rank.id} {...rank} />
          ))}
      </main>
      <footer className="mt-8 text-center text-gray-600">
        <p>&copy; 2023 Realtime Ranking List</p>
      </footer>
    </div>
  );
};

export default App;
